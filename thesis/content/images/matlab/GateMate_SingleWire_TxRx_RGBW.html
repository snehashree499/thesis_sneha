function [Frame_Out, Frame_Valid] = RX_Core(Serial_In)
% Simple single-wire 64-bit RGBW receiver
%
% Ts   = 0.25 ms
% Tbit = 1 ms
% 4 samples per bit
%
% Frame:
% Idle HIGH
% Start bit LOW
% 64 data bits, MSB first
% Stop bit HIGH

persistent active
persistent sample_count
persistent bit_count
persistent shift_reg
persistent latched_frame
persistent previous_value

if isempty(active)
    active         = false;
    sample_count   = uint16(0);
    bit_count      = uint16(0);
    shift_reg      = uint64(0);
    latched_frame  = uint64(0);
    previous_value = 1;
end

%% Default outputs

% IMPORTANT:
% Output only the last COMPLETE frame.
% Do not output the partially received shift register.
Frame_Out   = latched_frame;
Frame_Valid = false;

%% IDLE STATE

if ~active

    % Detect falling edge:
    % Idle HIGH -> Start bit LOW
    if (previous_value > 0.5) && (Serial_In < 0.5)

        active       = true;
        sample_count = uint16(0);
        bit_count    = uint16(0);
        shift_reg    = uint64(0);

    end

%% RECEIVE STATE

else

    sample_count = sample_count + 1;

    % Sample approximately at the centre of every data bit.
    %
    % Start bit occupies approximately samples 0...3.
    % First data-bit sample occurs at sample 6.
    % Then sample every 4 samples.

    if sample_count >= 6 && ...
       mod(double(sample_count - 6), 4) == 0 && ...
       bit_count < 64

        % Shift previous bits left
        shift_reg = bitshift(shift_reg, 1);

        % Insert new bit into LSB
        if Serial_In > 0.5
            shift_reg = bitor(shift_reg, uint64(1));
        end

        bit_count = bit_count + 1;
    end

    %% Complete frame received

    if bit_count == 64

        % Latch the COMPLETE frame
        latched_frame = shift_reg;

        % Update output
        Frame_Out = latched_frame;

        % One-sample valid pulse
        Frame_Valid = true;

        % Return receiver to idle
        active       = false;
        sample_count = uint16(0);
        bit_count    = uint16(0);

    end
end

previous_value = Serial_In;

end
